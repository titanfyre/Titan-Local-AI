cmake_minimum_required(VERSION 3.24)
project(TitanLocalAI VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(CTest)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Quick QuickControls2 Network Concurrent)
qt_standard_project_setup(REQUIRES 6.8)
add_library(browser_control_sdk STATIC third_party/browser_control/BrowserControl.cpp)
target_include_directories(browser_control_sdk PUBLIC third_party/browser_control third_party)
target_compile_definitions(browser_control_sdk PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
qt_add_executable(TitanLocalAI WIN32
    src/main.cpp src/AppController.h src/AppController.cpp
    src/ChatModel.h src/ChatModel.cpp src/SettingsStore.h src/SettingsStore.cpp
    src/LocalApiClient.h src/LocalApiClient.cpp src/AgentEngine.h src/AgentEngine.cpp
    src/ToolBroker.h src/ToolBroker.cpp src/BrowserTools.h src/BrowserTools.cpp
    src/runtime/ModelCatalog.h src/runtime/ModelCatalog.cpp
    src/runtime/RuntimeManager.h src/runtime/RuntimeManager.cpp
    src/tools/WorkspaceAccess.h src/tools/WorkspaceAccess.cpp)
target_link_libraries(TitanLocalAI PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Quick
    Qt6::QuickControls2 Qt6::Network Qt6::Concurrent browser_control_sdk dwmapi crypt32)
target_compile_definitions(TitanLocalAI PRIVATE TITAN_VERSION="${PROJECT_VERSION}" NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
    target_compile_options(TitanLocalAI PRIVATE /W4 /permissive- /utf-8 /EHsc)
    target_compile_options(browser_control_sdk PRIVATE /W4 /utf-8 /EHsc)
endif()
file(GLOB_RECURSE TITAN_QML CONFIGURE_DEPENDS RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} qml/*.qml)
foreach(qml_file IN LISTS TITAN_QML)
    get_filename_component(qml_name ${qml_file} NAME)
    set_source_files_properties(${qml_file} PROPERTIES QT_RESOURCE_ALIAS ${qml_name})
endforeach()
qt_add_qml_module(TitanLocalAI URI TitanLocalAI VERSION 1.0 QML_FILES ${TITAN_QML})
qt_add_resources(TitanLocalAI app_assets PREFIX "/" FILES assets/app-icon.ico)
if(WIN32)
    target_sources(TitanLocalAI PRIVATE assets/app.rc)
endif()
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(WorkspaceAccessTests tests/WorkspaceAccessTests.cpp src/tools/WorkspaceAccess.cpp src/tools/WorkspaceAccess.h)
    target_include_directories(WorkspaceAccessTests PRIVATE src)
    target_link_libraries(WorkspaceAccessTests PRIVATE Qt6::Core Qt6::Test)
    target_compile_definitions(WorkspaceAccessTests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    if(MSVC)
        target_compile_options(WorkspaceAccessTests PRIVATE /utf-8)
    endif()
    add_test(NAME WorkspaceAccess COMMAND WorkspaceAccessTests)
    qt_add_executable(AgentProtocolTests tests/AgentProtocolTests.cpp src/LocalApiClient.cpp src/LocalApiClient.h)
    target_include_directories(AgentProtocolTests PRIVATE src)
    target_link_libraries(AgentProtocolTests PRIVATE Qt6::Core Qt6::Network Qt6::Test)
    add_test(NAME AgentProtocol COMMAND AgentProtocolTests)
    qt_add_executable(RuntimeTests tests/RuntimeTests.cpp src/runtime/ModelCatalog.cpp src/runtime/ModelCatalog.h src/runtime/RuntimeManager.cpp src/runtime/RuntimeManager.h src/LocalApiClient.cpp src/LocalApiClient.h)
    target_include_directories(RuntimeTests PRIVATE src)
    target_link_libraries(RuntimeTests PRIVATE Qt6::Core Qt6::Network Qt6::Concurrent Qt6::Test)
    target_compile_definitions(RuntimeTests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    if(MSVC)
        target_compile_options(RuntimeTests PRIVATE /utf-8)
        target_compile_options(AgentProtocolTests PRIVATE /utf-8)
    endif()
    add_test(NAME Runtime COMMAND RuntimeTests)
    qt_add_executable(BrowserToolsTests tests/BrowserToolsTests.cpp src/BrowserTools.cpp src/BrowserTools.h)
    target_include_directories(BrowserToolsTests PRIVATE src)
    target_link_libraries(BrowserToolsTests PRIVATE Qt6::Core Qt6::Network Qt6::Concurrent Qt6::Test browser_control_sdk)
    if(MSVC)
        target_compile_options(BrowserToolsTests PRIVATE /utf-8)
    endif()
    add_test(NAME BrowserTools COMMAND BrowserToolsTests)
    set_tests_properties(BrowserTools PROPERTIES TIMEOUT 150)
    qt_add_executable(AgentEngineTests tests/AgentEngineTests.cpp src/AgentEngine.cpp src/AgentEngine.h src/ChatModel.cpp src/ChatModel.h src/LocalApiClient.cpp src/LocalApiClient.h src/ToolBroker.cpp src/ToolBroker.h src/BrowserTools.cpp src/BrowserTools.h src/tools/WorkspaceAccess.cpp src/tools/WorkspaceAccess.h)
    target_include_directories(AgentEngineTests PRIVATE src)
    target_link_libraries(AgentEngineTests PRIVATE Qt6::Core Qt6::Network Qt6::Concurrent Qt6::Test browser_control_sdk)
    target_compile_definitions(AgentEngineTests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    if(MSVC)
        target_compile_options(AgentEngineTests PRIVATE /utf-8)
    endif()
    add_test(NAME AgentEngine COMMAND AgentEngineTests)
    set_tests_properties(AgentEngine PROPERTIES TIMEOUT 90)
endif()
install(TARGETS TitanLocalAI RUNTIME DESTINATION .)
